create table Events(id NUM, title STR, location STR);
create table Locations(name STR, capacity NUM);
create table Participants(regNo NUM, event STR);
open table Events;
insert into Events values from events.csv;
open table Locations;
insert into Locations values from locations.csv;
open table Participants;
insert into Participants values from participants.csv;
select name from Locations join Events into EventLocations where Locations.name = Events.location;
open table Events;
select * from Events into AudiEvents where location = Audi;
open table AudiEvents;
select regNo, title from AudiEvents join Participants into AudiPeople where AudiEvents.title = Participants.event;
select * from Locations into LocationsGT100 where capacity >= 100;
open table LocationsGT100;
select * from LocationsGT100 into SmallLocations where capacity <= 200;
open table SmallLocations;
select * from SmallLocations join Events into SmallLocEvents where SmallLocations.name = Events.location;
open table SmallLocEvents;
select regNo, title from SmallLocEvents join Participants into MiniEventPeople where SmallLocEvents.title = Participants.event;
